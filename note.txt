with version_group as(
    select vehicle_id,
    case 
    when release_version < '2024.10' then 'old'
    else 'new'
    end as version_group
    from vehicles
),
trip_miles as(
    select
    vehicle_id,
    sum(miles) as miles
    from trips
    group by vehicle_id
),
diseng_cnt as(
    select
    t.vehicle_id,
    count(*) as diseng
    from trips t
    left join disengagement d
    on t.trip_id = d.trip_id
    group by t.vehicle_id
),
vehicle_metrics as(
    select 
    v.version_group,
    sum(diseng) as total_diseng,
    sum(miles) as total_miles
    from version_group v
    join trip_miles tm using(vehicle_id)
    join diseng_cnt dc using(vehicle_id)
    group by version_group
)

select *,
total_diseng * 1000.0 / total_miles as disengage_rate_per_1000_miles
from vehicle_metrics
